import numpy as np
import cv2
import os
from tqdm.notebook import tqdm

MAX_FRAMES = 90
TARGET_SIZE = (64, 32)

def load_data_with_masking(df, base_path):
    X_data = []      # Tensore video (N, 90, 32, 64, 3)
    y_data = []      # Etichette
    masks = []       # Maschere booleane (N, 90)
    
    kept = 0
    skipped = 0
        
    for row in tqdm(df.itertuples(), total=len(df)):
        full_path = os.path.join(base_path, row.user, row.file)
        
        cap = cv2.VideoCapture(full_path)
        frames = []
        while True:
            ret, frame = cap.read()
            if not ret: break
            frame = cv2.resize(frame, TARGET_SIZE)
            frames.append(frame)
        cap.release()
        
        L = len(frames)
        
        if L > MAX_FRAMES:
            skipped += 1
            continue
            
        if L == 0:
            skipped += 1
            continue
            
        padded_video = np.zeros((MAX_FRAMES, TARGET_SIZE[1], TARGET_SIZE[0], 3), dtype=np.uint8)
        
        mask = np.zeros((MAX_FRAMES,), dtype=np.float32)
        padded_video[:L] = np.array(frames)
        
        mask[:L] = 1.0
        
        X_data.append(padded_video)
        masks.append(mask)
        y_data.append(row.word)
        kept += 1

    print(f"\n✅ Completato!")
    print(f"📹 Video mantenuti: {kept}")
    print(f"🗑️ Video scartati (> {MAX_FRAMES} frames o errori): {skipped}")
    
    return np.array(X_data), np.array(y_data), np.array(masks)

import tensorflow as tf
from tensorflow.keras import layers, models, Input
from tensorflow.keras.callbacks import EarlyStopping, ReduceLROnPlateau
import matplotlib.pyplot as plt
import time

tf.keras.backend.clear_session()

INPUT_SHAPE = (90, 32, 64, 3) 
NUM_CLASSES = 71
BATCH_SIZE = 8 

def build_transfer_model():
    # INPUT
    video_input = Input(shape=INPUT_SHAPE, name='video_input')
    mask_input = Input(shape=(90,), name='mask_input') 
    
    base_cnn = tf.keras.applications.MobileNetV2(
        weights='imagenet', 
        include_top=False, 
        input_shape=(32, 64, 3)
    )
    
    base_cnn.trainable = False
    
    x = layers.TimeDistributed(base_cnn)(video_input)
    
    x = layers.TimeDistributed(layers.GlobalAveragePooling2D())(x)
    

    mask_reshaped = layers.Reshape((90, 1))(mask_input)
    x = layers.Multiply()([x, mask_reshaped])
    
    x = layers.Masking(mask_value=0.0)(x)
    x = layers.Bidirectional(layers.LSTM(128, return_sequences=False, dropout=0.3))(x)
    
    x = layers.Dense(128, activation='relu')(x)
    x = layers.Dropout(0.4)(x)
    output = layers.Dense(NUM_CLASSES, activation='softmax')(x)
    
    model = models.Model(inputs=[video_input, mask_input], outputs=output)
    return model

strategy = tf.distribute.MirroredStrategy()

with strategy.scope():
    model = build_transfer_model()
    model.compile(optimizer=tf.keras.optimizers.Adam(learning_rate=0.0005), 
                  loss='categorical_crossentropy',
                  metrics=['accuracy'])

model.summary()

start_time = time.time()

early_stop = EarlyStopping(monitor='val_loss', patience=10, restore_best_weights=True, verbose=1)
reduce_lr = ReduceLROnPlateau(monitor='val_loss', factor=0.5, patience=3, verbose=1)

history = model.fit(
    x=[X_train, mask_train], 
    y=y_train_cat,
    validation_data=([X_test, mask_test], y_test_cat),
    epochs=40,
    batch_size=BATCH_SIZE,
    callbacks=[early_stop, reduce_lr],
    verbose=1
)

training_time = time.time() - start_time
print(f"\n Tempo totale di training: {training_time/60:.2f} minuti")

loss, acc = model.evaluate([X_test, mask_test], y_test_cat)
print(f"\n TEST ACCURACY (Transfer Learning): {acc*100:.2f}%")

import tensorflow as tf
import numpy as np
import cv2

def get_video_saliency_map(model, video_sample, mask_sample, class_index):
    video_tensor = tf.convert_to_tensor(video_sample)
    mask_tensor = tf.convert_to_tensor(mask_sample)
    if len(video_tensor.shape) == 4:
        video_tensor = tf.expand_dims(video_tensor, axis=0)
        mask_tensor = tf.expand_dims(mask_tensor, axis=0)

    with tf.GradientTape() as tape:
        tape.watch(video_tensor)
        preds = model([video_tensor, mask_tensor], training=False)
        loss = preds[:, class_index]

    grads = tape.gradient(loss, video_tensor)
    if grads is None: return np.zeros(video_tensor.shape[1:-1])

    dgrad_abs = tf.math.abs(grads)
    heatmap = tf.reduce_max(dgrad_abs, axis=-1)[0] # (90, 32, 64)

    # --- MIGLIORAMENTO VISIVO ---
    heatmap = heatmap.numpy()

    v_max = np.percentile(heatmap, 99) 
    heatmap = np.clip(heatmap, 0, v_max)

    # Normalizzazione
    heatmap = (heatmap - np.min(heatmap)) / (v_max - np.min(heatmap) + 1e-8)

    # C. Smoothing (Sfocatura Gaussiana)
    # Questo unisce i puntini sparsi in una "chiazza" rossa più comprensibile
    # Applichiamo la sfocatura frame per frame
    heatmap_smoothed = []
    for frame_map in heatmap:
        # (5, 5) è la dimensione del filtro, aumentala se vuoi macchie più grandi
        blurred = cv2.GaussianBlur(frame_map, (5, 5), 0) 
        heatmap_smoothed.append(blurred)
    
    return np.array(heatmap_smoothed)

def plot_top_frames_gradcam(video, mask, model, class_names=None, num_frames_to_show=5):
    input_video = np.expand_dims(video, axis=0)
    input_mask = np.expand_dims(mask, axis=0)
    preds = model.predict([input_video, input_mask], verbose=0)
    predicted_idx = np.argmax(preds)
    confidence = np.max(preds)

    label_text = f"Classe {predicted_idx}"
    if class_names is not None and predicted_idx < len(class_names):
        label_text = class_names[predicted_idx]

    # 2. Calcolo Heatmap
    heatmap_video = get_video_saliency_map(model, video, mask, predicted_idx)

    # 3. Selezione frame migliori (somma dell'attivazione)
    frame_importance = np.sum(heatmap_video, axis=(1, 2))
    top_frame_indices = np.argsort(frame_importance)[-num_frames_to_show:][::-1]
    top_frame_indices = sorted(top_frame_indices) # Ordine cronologico

    # 4. Plotting
    plt.figure(figsize=(20, 5))
    for i, frame_idx in enumerate(top_frame_indices):
        ax = plt.subplot(1, num_frames_to_show, i + 1)
        
        # Preparazione immagine
        img = video[frame_idx].copy()
        
        # Preparazione Heatmap
        map_img = heatmap_video[frame_idx]
        map_img = cv2.resize(map_img, (img.shape[1], img.shape[0]))
        map_img = np.uint8(255 * map_img)
        heatmap_colored = cv2.applyColorMap(map_img, cv2.COLORMAP_JET)
        heatmap_colored = cv2.cvtColor(heatmap_colored, cv2.COLOR_BGR2RGB) / 255.0

        # Sovrapposizione
        superimposed_img = heatmap_colored * 0.4 + img * 0.6
        
        plt.imshow(superimposed_img)
        plt.title(f"Frame {frame_idx}\nImportanza: {frame_importance[frame_idx]:.1f}")
        plt.axis('off')

    plt.tight_layout()
    plt.show()
