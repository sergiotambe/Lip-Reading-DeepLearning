TRAIN_RATIO = 0.70
VAL_RATIO = 0.15
TEST_RATIO = 0.15

list_train = []
list_val = []
list_test = []

skipped_users = []

for user, group in df.groupby('user'):
    
    unique_words = group['word'].unique()
    n_words = len(unique_words)
    
    if n_words < 3:
        list_train.append(group)
        skipped_users.append(f"{user} ({n_words} parole)")
        continue

    np.random.seed(42) 
    shuffled_words = np.copy(unique_words)
    np.random.shuffle(shuffled_words)


    n_val = int(n_words * VAL_RATIO)
    n_test = int(n_words * TEST_RATIO)
    
    if n_val == 0: n_val = 1
    if n_test == 0: n_test = 1
    
    n_train = n_words - n_val - n_test
    
    words_val = shuffled_words[:n_val]
    words_test = shuffled_words[n_val : n_val + n_test]
    words_train = shuffled_words[n_val + n_test:]
    
    df_u_train = group[group['word'].isin(words_train)]
    df_u_val = group[group['word'].isin(words_val)]

import numpy as np
import cv2
import os
from tqdm.notebook import tqdm

MAX_FRAMES = 90
TARGET_SIZE = (64, 32)

"""
Carica i video, applica Zero Padding fino a MAX_FRAMES,
crea le maschere ed ESCLUDE i video che superano MAX_FRAMES.
"""

def load_data_with_masking(df, base_path):

    X_data = []      # Tensore video (N, 90, 32, 64, 3)
    y_data = []      # Etichette
    masks = []       # Maschere booleane (N, 90)
    
    kept = 0
    skipped = 0
        
    for row in tqdm(df.itertuples(), total=len(df)):
        full_path = os.path.join(base_path, row.user, row.file)
        
        cap = cv2.VideoCapture(full_path)
        frames = []
        while True:
            ret, frame = cap.read()
            if not ret: break
            frame = cv2.resize(frame, TARGET_SIZE)
            frames.append(frame)
        cap.release()
        
        L = len(frames)
        
        # Se è troppo lungo, lo scartiamo
        if L > MAX_FRAMES:
            skipped += 1
            continue
            
        if L == 0:
            skipped += 1
            continue
            
        padded_video = np.zeros((MAX_FRAMES, TARGET_SIZE[1], TARGET_SIZE[0], 3), dtype=np.uint8)
        
        mask = np.zeros((MAX_FRAMES,), dtype=np.float32)
        
        padded_video[:L] = np.array(frames)
        
        mask[:L] = 1.0
        
        X_data.append(padded_video)
        masks.append(mask)
        y_data.append(row.word)
        kept += 1

    print(f"\n✅ Completato!")
    print(f"📹 Video mantenuti: {kept}")
    print(f"🗑️ Video scartati (> {MAX_FRAMES} frames o errori): {skipped}")
    
    return np.array(X_data), np.array(y_data), np.array(masks)

INPUT_SHAPE = (90, 32, 64, 3) 
NUM_CLASSES = y_train_cat.shape[1]
BATCH_SIZE = 32

def build_transfer_model_v2():
    video_input = Input(shape=INPUT_SHAPE, name='video_input')
    mask_input = Input(shape=(90,), name='mask_input') 
    
    # MobileNetV2 vuole dati tra [-1, 1].
    x = layers.TimeDistributed(layers.Rescaling(scale=2.0, offset=-1.0))(video_input)
    
    # rete pre-addestrata (senza la testa)
    base_cnn = tf.keras.applications.MobileNetV2(
        weights='imagenet', 
        include_top=False, 
        input_shape=(32, 64, 3)
    )
    
    # fase 1 congeliamo tutto
    base_cnn.trainable = False
    
    x = layers.TimeDistributed(base_cnn, name='frozen_mobilenet')(x)
    
    x = layers.TimeDistributed(layers.GlobalAveragePooling2D())(x)
    
    # --- MASKING & LSTM ---
    mask_reshaped = layers.Reshape((90, 1))(mask_input)
    x = layers.Multiply()([x, mask_reshaped])
    
    x = layers.Masking(mask_value=0.0)(x)
    x = layers.Bidirectional(layers.LSTM(128, return_sequences=False, dropout=0.3))(x)
    
    x = layers.Dense(128, activation='relu')(x)
    x = layers.Dropout(0.4)(x)
    output = layers.Dense(NUM_CLASSES, activation='softmax')(x)
    
    model = models.Model(inputs=[video_input, mask_input], outputs=output)
    return model
